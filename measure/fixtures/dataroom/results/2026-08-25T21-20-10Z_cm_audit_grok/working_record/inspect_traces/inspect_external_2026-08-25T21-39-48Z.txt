139|def _apply_row_scope(
140|    statement: exp.Select,
141|    scoped: dict,
142|    scope_value: Optional[str],
143|    cte_names: set,
144|    case_insensitive: bool = False,
145|) -> tuple:
146|    """Rewrite every scoped base table into a pre-filtered subquery:
147|
148|        FROM orders o  ->  FROM (SELECT * FROM orders WHERE email = 'x') AS o
149|
150|    Filtering the *relation* rather than checking the model's WHERE clause is
151|    what makes this unbypassable. Whatever the model writes outside — OR 1=1,
152|    a UNION arm, a correlated subquery — it can only ever see rows the wrapper
153|    already returned. Validating a model-supplied predicate instead would mean
154|    proving that no OR, no join and no outer reference widens it, which is not
155|    a thing you can check reliably.
156|
157|    Returns (statement, error). Fails closed: a scoped table with no value to
158|    scope by is refused, never silently unscoped.
159|    """
160|    targets = []
161|    for table in statement.find_all(exp.Table):
162|        reference = _table_reference(table)
163|        if not reference:
164|            continue
165|        if not table.db and reference in cte_names:
166|            continue
167|        column = scoped.get(reference)
168|        if column is None and table.name:
169|            column = scoped.get(table.name.lower())
170|        if column:
171|            targets.append((table, column))
172|
173|    if not targets:
174|        return statement, None
175|
176|    if not (scope_value or "").strip():
177|        return None, (
178|            "This table is restricted to the rows of the ticket's own customer, "
179|            "and this ticket has no customer identity to scope by. Set the "
180|            "ticket's customer, or query a table that is not customer-scoped."
181|        )
182|
183|    value = scope_value.strip()
184|    for table, column in targets:
185|        alias_name = table.alias or table.name
186|        inner_table = table.copy()
187|        inner_table.set("alias", None)
188|        column_ref = exp.Column(this=exp.to_identifier(column))
189|        # Emails are compared case-insensitively — the same person may be
190|        # stored as Bob@x.com here and bob@x.com there, and an exact match
191|        # would silently return nothing rather than their rows. Costs an index
192|        # on the raw column unless one exists on LOWER(col).
193|        predicate = exp.EQ(
194|            this=exp.Lower(this=column_ref) if case_insensitive else column_ref,
195|            # Literal.string escapes on render — never string-format a value
196|            # into SQL.
197|            expression=exp.Literal.string(value.lower() if case_insensitive else value),
198|        )
199|        inner = (
200|            exp.Select().select(exp.Star()).from_(inner_table).where(predicate)
201|        )
202|        table.replace(
203|            exp.Subquery(this=inner, alias=exp.TableAlias(this=exp.to_identifier(alias_name)))
204|        )
205|    return statement, None
