132|def is_encrypted(value: Any) -> bool:

137|def encrypt_value(value: Optional[str]) -> Optional[str]:
138|    """Encrypt a string for storage. Idempotent: already-encrypted input is returned

140|    if value is None:
141|        return None

148|        token = fernet.encrypt(str(value).encode()).decode()

155|def decrypt_value(value: Optional[str]) -> Optional[str]:
156|    """Decrypt a stored string.
157|
158|    Values without the ``enc:`` prefix are returned unchanged — that is a row the
159|    backfill has not reached yet, and reads must keep working throughout the
160|    migration. A key we no longer hold, or a corrupt token, logs and returns None
161|    rather than raising: one damaged row must not fail the whole request.
162|    """
163|    if not is_encrypted(value):
164|        return value
165|
166|    parts = value.split(SEPARATOR, 2)
167|    if len(parts) != 3:
168|        logger.error("Malformed ciphertext: missing key id or token")
169|        return None

175|        return None

178|        return fernet.decrypt(token.encode()).decode()
